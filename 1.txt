namespace AmpPins {
constexpr int LRC = 4;
constexpr int BCLK = 5;
constexpr int DIN = 6;
constexpr int GAIN = 7;
constexpr int SD = 15;
}

namespace DisplayPins {
constexpr int SCLK = 20;
constexpr int MOSI = 21;
constexpr int RST = 47;
constexpr int DC = 48;
constexpr int CS = 45;
constexpr int BLK = 35;
}

struct ButtonPair {
    uint8_t inputPin;
    uint8_t lowPin;
    bool rawState;
    bool stableState;
    uint32_t changedAt;
};

ButtonPair buttons[] = {
    {17, 18, false, false, 0},
    {8, 40, false, false, 0},
    {46, 9, false, false, 0},
};

constexpr uint8_t BUTTON_COUNT = 3;
constexpr uint8_t VOICE_COUNT = 4;
constexpr uint8_t BUFFER_COUNT = VOICE_COUNT + 2;
constexpr uint16_t AUDIO_FRAMES = 64;
constexpr uint32_t SAMPLE_RATE = 16000;
constexpr uint32_t MAX_SAMPLE_BYTES = 160 * 1024;
constexpr uint32_t RELEASE_DEBOUNCE_MS = 3;
constexpr bool AUTO_PLAY = false;
constexpr uint16_t SONG_BPM = 119;
constexpr uint8_t STEPS_PER_BEAT = 2; // Eighth-note score steps.
constexpr uint32_t AUTO_NOTE_INTERVAL_US = 60000000UL / (SONG_BPM * STEPS_PER_BEAT);
constexpr i2s_port_t I2S_PORT = I2S_NUM_0;

constexpr uint16_t C_BLACK = 0x0000;
constexpr uint16_t C_WHITE = 0xFFFF;
constexpr uint16_t C_CYAN = 0x07FF;
constexpr uint16_t C_BLUE = 0x0255;
constexpr uint16_t C_GREEN = 0x07E0;
constexpr uint16_t C_RED = 0xF800;
constexpr uint16_t C_YELLOW = 0xFFE0;
constexpr uint16_t C_DARK = 0x18C3;

Arduino_DataBus *displayBus = new Arduino_ESP32SPI(
    DisplayPins::DC, DisplayPins::CS, DisplayPins::SCLK, DisplayPins::MOSI,
    GFX_NOT_DEFINED, FSPI, false);
Arduino_GFX *display = new Arduino_NV3007(
    displayBus, DisplayPins::RST, 0, true, 142, 428,
    12, 0, 14, 0,
    nv3007_279_init_operations, sizeof(nv3007_279_init_operations));
bool displayReady = false;

struct LoadedNote {
    int16_t *samples = nullptr;
    uint32_t frameCount = 0;
};

struct MemoryVoice {
    int16_t *samples = nullptr;
    uint32_t frameCount = 0;
    uint32_t position = 0;
    bool active = false;
};

MemoryVoice voices[VOICE_COUNT];
QueueHandle_t noteRequestQueue = nullptr;
QueueHandle_t loadedNoteQueue = nullptr;
QueueHandle_t freeBufferQueue = nullptr;
QueueHandle_t triggerQueue = nullptr;
uint8_t nextVoice = 0;
uint16_t prefetchStep = 0;
uint16_t playbackStep = 0;
uint32_t nextAutoNoteAtUs = 0;
uint8_t expectedButton = 0;
int8_t lastPressedButton = -1;
int16_t lastPlayedSample = -1;
bool lastPressCorrect = false;

void noteName(uint8_t sampleIndex, char *buffer, size_t length) {
    static const char *names[] = {"C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"};
    const uint8_t midi = 21 + sampleIndex;
    snprintf(buffer, length, "%s%d", names[midi % 12], static_cast<int>(midi / 12) - 1);
}

void panelText(const char *text, int16_t left, int16_t right, int16_t y,
               uint8_t size, uint16_t color) {
    int16_t x = left + ((right - left) - static_cast<int16_t>(strlen(text) * 6 * size)) / 2;
    if (x < left) x = left;
    display->setTextSize(size);
    display->setTextColor(color);
    display->setCursor(x, y);
    display->print(text);
}

void drawInterface() {
    if (!displayReady) return;
    display->fillScreen(C_BLACK);
    display->fillRect(0, 0, display->width(), 24, C_BLUE);
    panelText("LOVE STORY", 0, 150, 5, 2, C_WHITE);
    panelText("3-KEY DEBUG", 150, 428, 8, 1, C_CYAN);
    display->drawFastVLine(140, 28, 90, C_DARK);
    display->drawFastVLine(290, 28, 90, C_DARK);

    char line[32];
    snprintf(line, sizeof(line), "STEP %03u/%03u", playbackStep + 1, SCORE_LENGTH);
    panelText(line, 0, 140, 36, 1, C_YELLOW);
    panelText("NEXT NOTE", 0, 140, 58, 1, C_CYAN);
    char note[8];
    noteName(pgm_read_byte(&LOVE_STORY_SCORE[playbackStep]), note, sizeof(note));
    panelText(note, 0, 140, 78, 3, C_WHITE);

    panelText("NEXT KEY", 140, 290, 36, 2, C_CYAN);
    snprintf(line, sizeof(line), "%u", expectedButton + 1);
    panelText(line, 140, 290, 65, 5, C_YELLOW);

    if (lastPressedButton < 0) {
        panelText("READY", 290, 428, 52, 2, C_GREEN);
        panelText("PRESS TARGET", 290, 428, 82, 1, C_WHITE);
    } else {
        snprintf(line, sizeof(line), "PRESSED %u", lastPressedButton + 1);
        panelText(line, 290, 428, 42, 1, C_CYAN);
        panelText(lastPressCorrect ? "CORRECT" : "WRONG", 290, 428, 64, 2,
                  lastPressCorrect ? C_GREEN : C_RED);
        if (lastPlayedSample >= 0) {
            noteName(static_cast<uint8_t>(lastPlayedSample), note, sizeof(note));
            panelText(note, 290, 428, 92, 2, C_YELLOW);
        } else {
            panelText("NO SOUND", 290, 428, 92, 1, C_RED);
        }
    }

    display->fillRect(0, 124, display->width(), 18, C_DARK);
    panelText("K1 17/18   K2 8/40   K3 46/9", 0, 428, 130, 1, C_WHITE);
}

void assignRandomTarget() {
    expectedButton = static_cast<uint8_t>(random(BUTTON_COUNT));
    drawInterface();
}

uint32_t readLe32(const uint8_t *bytes) {
    return static_cast<uint32_t>(bytes[0])
        | (static_cast<uint32_t>(bytes[1]) << 8)
        | (static_cast<uint32_t>(bytes[2]) << 16)
        | (static_cast<uint32_t>(bytes[3]) << 24);
}

void returnBuffer(int16_t *buffer) {
    if (buffer != nullptr) xQueueSend(freeBufferQueue, &buffer, portMAX_DELAY);
}

void stopVoice(MemoryVoice &voice) {
    if (voice.samples != nullptr) returnBuffer(voice.samples);
    voice.samples = nullptr;
    voice.frameCount = 0;
    voice.position = 0;
    voice.active = false;
}

bool loadWavIntoBuffer(uint8_t sample, int16_t *buffer, LoadedNote &loaded) {
    char path[24];
    snprintf(path, sizeof(path), "/piano/%02u.wav", sample + 1);
    File file = SPIFFS.open(path, "r");
    if (!file) return false;

    uint8_t riffHeader[12];
    if (file.read(riffHeader, sizeof(riffHeader)) != sizeof(riffHeader)
        || memcmp(riffHeader, "RIFF", 4) != 0
        || memcmp(riffHeader + 8, "WAVE", 4) != 0) {
        file.close();
        return false;
    }

    uint32_t dataSize = 0;
    while (file.available()) {
        uint8_t chunkHeader[8];
        if (file.read(chunkHeader, sizeof(chunkHeader)) != sizeof(chunkHeader)) break;
        const uint32_t chunkSize = readLe32(chunkHeader + 4);
        if (memcmp(chunkHeader, "data", 4) == 0) {
            dataSize = chunkSize;
            break;
        }
        if (!file.seek(file.position() + chunkSize + (chunkSize & 1U))) break;
    }

    if (dataSize == 0 || dataSize > MAX_SAMPLE_BYTES) {
        file.close();
        return false;
    }

    uint32_t totalRead = 0;
    uint8_t *destination = reinterpret_cast<uint8_t *>(buffer);
    while (totalRead < dataSize) {
        const size_t block = min(static_cast<uint32_t>(4096), dataSize - totalRead);
        const size_t readNow = file.read(destination + totalRead, block);
        if (readNow == 0) break;
        totalRead += readNow;
    }
    file.close();
    if (totalRead != dataSize) return false;

    loaded.samples = buffer;
    loaded.frameCount = dataSize / sizeof(int16_t);
    return true;
}

void loaderTask(void *) {
    while (true) {
        uint8_t requestedSample = 0;
        xQueueReceive(noteRequestQueue, &requestedSample, portMAX_DELAY);

        int16_t *buffer = nullptr;
        xQueueReceive(freeBufferQueue, &buffer, portMAX_DELAY);
        LoadedNote loaded;
        if (loadWavIntoBuffer(requestedSample, buffer, loaded)) {
            xQueueSend(loadedNoteQueue, &loaded, portMAX_DELAY);
        } else {
            returnBuffer(buffer);
        }
    }
}

void audioTask(void *) {
    int32_t mix[AUDIO_FRAMES];
    int16_t stereo[AUDIO_FRAMES * 2];

    while (true) {
        uint8_t trigger = 0;
        while (xQueueReceive(triggerQueue, &trigger, 0) == pdTRUE) {
            LoadedNote loaded;
            if (xQueueReceive(loadedNoteQueue, &loaded, 0) == pdTRUE) {
                MemoryVoice &voice = voices[nextVoice];
                nextVoice = (nextVoice + 1) % VOICE_COUNT;
                stopVoice(voice);
                voice.samples = loaded.samples;
                voice.frameCount = loaded.frameCount;
                voice.position = 0;
                voice.active = true;
            }
        }

        memset(mix, 0, sizeof(mix));
        for (uint8_t voiceIndex = 0; voiceIndex < VOICE_COUNT; ++voiceIndex) {
            MemoryVoice &voice = voices[voiceIndex];
            if (!voice.active) continue;

            const uint32_t available = voice.frameCount - voice.position;
            const uint16_t frames = min(static_cast<uint32_t>(AUDIO_FRAMES), available);
            for (uint16_t frame = 0; frame < frames; ++frame) {
                mix[frame] += voice.samples[voice.position + frame];
            }
            voice.position += frames;
            if (voice.position >= voice.frameCount) stopVoice(voice);
        }

        for (uint16_t frame = 0; frame < AUDIO_FRAMES; ++frame) {
            int32_t value = mix[frame] / 2; // Master volume 0.50.
            if (value > 32767) value = 32767;
            if (value < -32768) value = -32768;
            stereo[frame * 2] = static_cast<int16_t>(value);
            stereo[frame * 2 + 1] = static_cast<int16_t>(value);
        }

        size_t bytesWritten = 0;
        i2s_write(I2S_PORT, stereo, sizeof(stereo), &bytesWritten, portMAX_DELAY);
    }
}

int8_t readNewPress() {
    const uint32_t now = millis();
    for (uint8_t i = 0; i < BUTTON_COUNT; ++i) {
        const bool pressed = digitalRead(buttons[i].inputPin) == LOW;
        if (pressed != buttons[i].rawState) {
            buttons[i].rawState = pressed;
            buttons[i].changedAt = now;
            if (pressed && !buttons[i].stableState) {
                buttons[i].stableState = true;
                return static_cast<int8_t>(i);
            }
        }
        if (!pressed && buttons[i].stableState && now - buttons[i].changedAt >= RELEASE_DEBOUNCE_MS) {
            buttons[i].stableState = false;
        }
    }
    return -1;
}

void prefetchNextScoreNote() {
    const uint8_t sample = pgm_read_byte(&LOVE_STORY_SCORE[prefetchStep]);
    if (xQueueSend(noteRequestQueue, &sample, 0) == pdTRUE) {
        ++prefetchStep;
        if (prefetchStep >= SCORE_LENGTH) prefetchStep = 0;
    }
}

bool triggerNextScoreNote() {
    if (uxQueueMessagesWaiting(loadedNoteQueue) == 0) return false;
    const uint8_t trigger = 1;
    if (xQueueSend(triggerQueue, &trigger, 0) == pdTRUE) {
        prefetchNextScoreNote();
        return true;
    }
    return false;
}

void setupI2S() {
    const i2s_config_t config = {
        .mode = static_cast<i2s_mode_t>(I2S_MODE_MASTER | I2S_MODE_TX),
        .sample_rate = SAMPLE_RATE,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
        .channel_format = I2S_CHANNEL_FMT_RIGHT_LEFT,
        .communication_format = I2S_COMM_FORMAT_STAND_I2S,
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 6,
        .dma_buf_len = AUDIO_FRAMES,
        .use_apll = false,
        .tx_desc_auto_clear = true,
        .fixed_mclk = 0,
        .mclk_multiple = I2S_MCLK_MULTIPLE_DEFAULT,
        .bits_per_chan = I2S_BITS_PER_CHAN_DEFAULT,
    };
    const i2s_pin_config_t pins = {
        .mck_io_num = I2S_PIN_NO_CHANGE,
        .bck_io_num = AmpPins::BCLK,
        .ws_io_num = AmpPins::LRC,
        .data_out_num = AmpPins::DIN,
        .data_in_num = I2S_PIN_NO_CHANGE,
    };
    ESP_ERROR_CHECK(i2s_driver_install(I2S_PORT, &config, 0, nullptr));
    ESP_ERROR_CHECK(i2s_set_pin(I2S_PORT, &pins));
    i2s_zero_dma_buffer(I2S_PORT);
}

void setup() {
    Serial.begin(115200);
    delay(500);

    pinMode(AmpPins::SD, OUTPUT);
    digitalWrite(AmpPins::SD, HIGH);
    pinMode(AmpPins::GAIN, INPUT);
    for (uint8_t i = 0; i < BUTTON_COUNT; ++i) {
        pinMode(buttons[i].lowPin, OUTPUT);
        digitalWrite(buttons[i].lowPin, LOW);
        pinMode(buttons[i].inputPin, INPUT_PULLUP);
    }
    pinMode(DisplayPins::BLK, OUTPUT);
    digitalWrite(DisplayPins::BLK, HIGH);

    displayReady = display->begin(40000000);
    if (displayReady) {
        display->setRotation(1);
        Serial.printf("DISPLAY ready: %d x %d (horizontal)\n",
                      display->width(), display->height());
    } else {
        Serial.println("NV3007 display initialization failed");
    }

    if (!SPIFFS.begin(false)) {
        Serial.println("SPIFFS mount failed");
        while (true) delay(1000);
    }
    if (!psramFound()) {
        Serial.println("PSRAM not found");
        while (true) delay(1000);
    }

    noteRequestQueue = xQueueCreate(16, sizeof(uint8_t));
    loadedNoteQueue = xQueueCreate(2, sizeof(LoadedNote));
    freeBufferQueue = xQueueCreate(BUFFER_COUNT, sizeof(int16_t *));
    triggerQueue = xQueueCreate(8, sizeof(uint8_t));
    if (noteRequestQueue == nullptr || loadedNoteQueue == nullptr
        || freeBufferQueue == nullptr || triggerQueue == nullptr) {
        Serial.println("Failed to create audio queues");
        while (true) delay(1000);
    }

    for (uint8_t i = 0; i < BUFFER_COUNT; ++i) {
        int16_t *buffer = static_cast<int16_t *>(
            heap_caps_malloc(MAX_SAMPLE_BYTES, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT));
        if (buffer == nullptr) {
            Serial.println("Failed to allocate PSRAM audio pool");
            while (true) delay(1000);
        }
        xQueueSend(freeBufferQueue, &buffer, portMAX_DELAY);
    }

    setupI2S();
    xTaskCreatePinnedToCore(audioTask, "pcm-audio", 4096, nullptr, 4, nullptr, 0);
    xTaskCreatePinnedToCore(loaderTask, "pcm-loader", 4096, nullptr, 2, nullptr, 1);
    prefetchNextScoreNote();
    prefetchNextScoreNote();
    randomSeed(esp_random());
    assignRandomTarget();
    nextAutoNoteAtUs = micros() + 750000;
    Serial.printf("Love Story 3-key horizontal UI; %u KB PSRAM; volume 0.50\n",
                  ESP.getPsramSize() / 1024);
    Serial.println("KEY1 input 17 / output-low 18; KEY2 input 8 / output-low 3; KEY3 input 46 / output-low 9");
}

void loop() {
    if (AUTO_PLAY) {
        const uint32_t now = micros();
        if (static_cast<int32_t>(now - nextAutoNoteAtUs) >= 0) {
            if (triggerNextScoreNote()) {
                ++playbackStep;
                if (playbackStep >= SCORE_LENGTH) playbackStep = 0;
                assignRandomTarget();
            }
            nextAutoNoteAtUs += AUTO_NOTE_INTERVAL_US;
            if (static_cast<int32_t>(now - nextAutoNoteAtUs) >= 0) {
                nextAutoNoteAtUs = now + AUTO_NOTE_INTERVAL_US;
            }
        }
        return;
    }
    const int8_t pressed = readNewPress();
    if (pressed >= 0) {
        lastPressedButton = pressed;
        Serial.printf("KEY%d pressed; expected KEY%d", pressed + 1, expectedButton + 1);
        if (static_cast<uint8_t>(pressed) == expectedButton) {
            lastPressCorrect = true;
            lastPlayedSample = pgm_read_byte(&LOVE_STORY_SCORE[playbackStep]);
            if (triggerNextScoreNote()) {
                Serial.printf("; CORRECT; note %d; step %u\n",
                              lastPlayedSample, static_cast<unsigned>(playbackStep + 1));
                ++playbackStep;
                if (playbackStep >= SCORE_LENGTH) playbackStep = 0;
                assignRandomTarget();
            } else {
                Serial.println("; CORRECT but audio buffer not ready");
                drawInterface();
            }
        } else {
            lastPressCorrect = false;
            lastPlayedSample = -1;
            Serial.println("; WRONG; silent");
            drawInterface();
        }
    }
}
